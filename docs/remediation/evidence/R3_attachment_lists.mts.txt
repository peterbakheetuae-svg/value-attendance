// R3 before/after: list timing, RSS, payload hash (synthetic va_perf, 500 listed rows each with a 200 KB attachment)
import { createHash } from 'node:crypto';
import { writeFileSync } from 'node:fs';
const ROOT = process.env.ROOT ?? '/home/claude/peterbakheetuae-svg/va-cf/api';
const { createDb } = await import(ROOT + '/src/db.ts');
const { buildRouter } = await import(ROOT + '/src/app.ts');
const { createFetchHandler } = await import(ROOT + '/src/http/router.ts');
const { systemClock } = await import(ROOT + '/src/lib/time.ts');
const label = process.argv[2] ?? 'run';
const db = createDb('postgres://postgres:dev@127.0.0.1:5433/va_perf', 3);
const h = createFetchHandler(buildRouter({ db, clock: systemClock }), { corsOrigins: [] });
const tok = (await (await h(new Request('http://x/auth/login', { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify({ identifier: 'OWNER', password: 'Passw0rd!x', deviceId: 'perf-OWNER00000000' }) }))).json()).token;
for (const path of ['/cash/receipts', '/cash/expenses']) {
  const times: number[] = []; let body = ''; let rss = 0;
  for (let i = 0; i < 5; i++) {
    const m0 = process.memoryUsage().rss; const t0 = performance.now();
    const r = await h(new Request('http://x' + path, { headers: { authorization: 'Bearer ' + tok } }));
    body = await r.text();
    times.push(performance.now() - t0); rss = Math.max(rss, process.memoryUsage().rss - m0);
    if (r.status !== 200) throw new Error(path + ' ' + r.status);
  }
  const n = (JSON.parse(body).receipts ?? JSON.parse(body).expenses).length;
  writeFileSync(`/tmp/r3_${label}${path.replace(/\//g, '_')}.json`, body);
  times.sort((a, b) => a - b);
  console.log(`${label} ${path} rows=${n} median=${times[2].toFixed(1)}ms min=${times[0].toFixed(1)}ms maxRSS+=${(rss / 1048576).toFixed(0)}MB payload=${(body.length / 1024).toFixed(0)}KB sha256=${createHash('sha256').update(body).digest('hex').slice(0, 16)}`);
}
await db.end();
